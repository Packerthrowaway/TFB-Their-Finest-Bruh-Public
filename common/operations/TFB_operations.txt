kill_heydrich = { ### 1/50
	icon = GFX_murder_trotsky 
	map_icon = GFX_murder_trotsky_map
	name = murder_heydrich
	desc = murder_heydrich_desc
	days = 90

	priority = 25

	allowed = {
		original_tag = ENG
	}
	available = {
		network_strength = {
			target = GER
			state = 9
			value > 25
		}
	}
	visible = {
		has_completed_focus = ENG_assasinate_heydrich
		FROM = {
			controls_state = 9
		}
		#GER = { has_character = GER_reinhard_heydrich  } Unless you retire him then this check doesnt work
		NOT = { has_global_flag = heydrich_killed }
		has_war_with = GER
	}

	operatives = 2
	# Integer [0-100] - Every operation has a hard requirement of network strength in the target country
	network_strength = 50
	equipment = {
		infantry_equipment = 2000
	}
	on_start = {  # SCOPE_OPERATION
		
	}
	outcome_modifiers = { operation_outcome }
	cost_modifiers = { operation_cost }
	risk_modifiers = { operation_risk }
	outcome_potential = {
		custom_effect_tooltip = kill_heydrich_outcome_tt
	}
	risk_chance = 0.4
	experience = 5

	# outcome is an effect that runs when the operation has completed
	outcome_execute = { # SCOPE_OPERATION
		ROOT = { 
			random_list = { 
				
				#100 = { GER = {  retire_character = GER_reinhard_heydrich } } Use this to kill him imo but flag could work
				25 = { country_event = tfb_kill_heydrich.1 } #raid succeeds
				50 = { country_event = tfb_kill_heydrich.2 } #heydrich wounded
				25 = { country_event = tfb_kill_heydrich.3 } #raid fails
			}
		}
	}
	
	# This is where all the separate phases are defined
	# Each phase for an operation may have a single actual phase assigned to it, which is determined here
	phases = { #infiltration 
		infiltration_border = { base = 60 }
		infiltration_diplomatic = {
			base = 20
			modifier = {
				FROM = {
					has_war_with = ROOT
				}
				factor = 0
			}
		}
	}
	phases = { 
		murder_trotsky_attack = { base = 100 }
	}
	phases = { 
		#exfiltration_border = { base = 40 }
		exfiltration_go_to_ground = { base = 30 }
		exfiltration_air_pickup = { 
			base = 10
			modifier = {
				ROOT = { 
					has_equipment = {
						transport_plane_equipment < 1
					}
				}
				factor = 0 #less likely if we have no transports
			}
		}
	}
}
